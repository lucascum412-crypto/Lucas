// --- 新增：碰撞檢測工具函數 --- function isColliding(rect1, rect2) { return rect1.x - rect1.w/2 < rect2.x + rect2.w/2 && rect1.x + rect1.w/2 > rect2.x - rect2.w/2 && rect1.y - rect1.h/2 < rect2.y + rect2.h/2 && rect1.y + rect1.h/2 > rect2.y - rect2.h/2; } // 獲取場景中所有實體障礙物 function getObstacles() { let obs = []; // 門口（作為障礙物） gates.forEach(g => obs.push({ x: g.x + 50, y: g.y + 65, w: 100, h: 130 })); // 裝飾物 decoList.forEach(d => { const rx = d.x * canvas.width, ry = d.y * canvas.height; if(d.type === 'bench') obs.push({ x: rx + 50, y: ry + 17, w: 110, h: 40 }); if(d.type === 'tree') obs.push({ x: rx, y: ry, w: 40, h: 40 }); if(d.type === 'lamp') obs.push({ x: rx + 3, y: ry + 45, w: 15, h: 90 }); }); // 其他 NPC (對玩家而言) npcs.forEach(n => obs.push({ x: n.x, y: n.y, w: n.w, h: n.h })); return obs; } function update() { let nextX = player.x; let nextY = player.y; // 1. 預算玩家位置 if (keys['ArrowUp'] || keys['w']) nextY -= player.speed; if (keys['ArrowDown'] || keys['s']) nextY += player.speed; if (keys['ArrowLeft'] || keys['a']) nextX -= player.speed; if (keys['ArrowRight'] || keys['d']) nextX += player.speed; // 2. 邊界檢查 nextX = Math.max(20, Math.min(canvas.width - 20, nextX)); nextY = Math.max(20, Math.min(canvas.height - 20, nextY)); // 3. 碰撞檢查 (玩家 vs 障礙物) let canMoveX = true; let canMoveY = true; const obstacles = getObstacles(); const tempPlayerX = { ...player, x: nextX, y: player.y }; const tempPlayerY = { ...player, x: player.x, y: nextY }; obstacles.forEach(ob => { if (isColliding(tempPlayerX, ob)) canMoveX = false; if (isColliding(tempPlayerY, ob)) canMoveY = false; }); if (canMoveX) player.x = nextX; if (canMoveY) player.y = nextY; // 4. NPC AI 與碰撞 npcs.forEach(n => { if (n.state === 'wait') { n.timer--; if (n.timer <= 0) { n.state = 'move'; n.tx = Math.random() * (canvas.width - 60) + 30; n.ty = Math.random() * (canvas.height - 60) + 30; } } else { let dx = n.tx - n.x, dy = n.ty - n.y, dist = Math.hypot(dx, dy); if (dist < 2) { n.state = 'wait'; n.timer = Math.random() * 300 + 100; } else { let nNextX = n.x + (dx/dist) * 1.2; let nNextY = n.y + (dy/dist) * 1.2; // NPC 簡單碰撞檢查 (不穿過玩家與其他障礙物) let blocked = false; const otherEntities = [player, ...npcs.filter(other => other !== n), ...getObstacles().filter(o => o.x !== n.x)]; otherEntities.forEach(ent => { if (isColliding({x: nNextX, y: nNextY, w: n.w, h: n.h}, ent)) blocked = true; }); if (!blocked) { n.x = nNextX; n.y = nNextY; } else { n.state = 'wait'; // 撞到東西就停下來思考 n.timer = 50; } } } }); // 傳送檢查 (維持原樣) // ... }
